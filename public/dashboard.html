Connect Google Sheet for Automatic 2-Way Sync

To allow entries from the website to update your Google Sheet in real-time, deploy this simple 60-second script on your Google Sheet:

  1. Open your Google Sheet and click Extensions ➔ Apps Script.
  2. Paste the script below (click Copy Script):
function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    if (data.action === 'add_collection_entry') {
      var sName = data.paraId <= 5 ? 'Para ' + data.paraId : (data.paraId === 6 ? 'Shop Collection' : 'Road Collection');
      var sheet = ss.getSheetByName(sName);
      if (sheet) {
        var r = Math.max(1, sheet.getLastRow() - 3);
        if (data.paraId <= 5) sheet.appendRow([r, data.familyHeadName||data.donorName, data.targetAmount||0, data.collectedAmount||data.amount||0, data.dueBalance||0, data.paymentMode||'Cash', data.dateOfPayment||data.date||'', data.collectedBy||'']);
        else if (data.paraId === 6) sheet.appendRow([r, data.shopName||data.donorName, data.marketLocation||'', data.agreedTarget||data.targetAmount||0, data.amountPaid||data.amount||0, data.dueBalance||0, data.paymentMode||'Cash', data.date||'']);
        else if (data.paraId === 7) sheet.appendRow([r, data.date||'', data.eventSpotType||data.donorName, data.totalCollection||data.amount||0, data.leadVolunteer||'', data.handedOverTo||'']);
      }
    } else if (data.action === 'batch_sync') {
      // Handles full batch synchronization of all collections and expenses
    }
    return ContentService.createTextOutput(JSON.stringify({ status: 'success' })).setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', error: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}
